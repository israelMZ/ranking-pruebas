function construirPizarra() { const container = document.getElementById('pizarraContainer'); if(!container) return; container.innerHTML = ''; let totalEstrellasGeneral = 0; let amClaro = 0, amWow = 0, amMovistarHogar = 0, amMovistarMovil = 0; let pmClaro = 0, pmWow = 0, pmMovistarHogar = 0, pmMovistarMovil = 0; EQUIPOS_DATA.forEach(eq => { const divSeccion = document.createElement('div'); divSeccion.className = `pizarra-seccion ${eq.color}`; // 1. Contar estrellas del propio TL const historialTL = estrellasData[eq.lider] || []; let estrellasEquipo = historialTL.length; // 2. Contar estrellas de los asesores del equipo eq.asesores.forEach(as => { const h = estrellasData[as] || []; estrellasEquipo += h.length; }); const metaEq = metasData[eq.id] !== undefined ? metasData[eq.id] : 7; const faltanEq = metaEq - estrellasEquipo; const txtFaltanEq = faltanEq > 0 ? `Faltan ${faltanEq}` : `¡COMPLETADO! 🎀`; // Generar la tarjeta del TL con sus propios botones de venta let html = `
👑 TL: ${eq.lider}
🎯 Objetivo: ${estrellasEquipo} / ${metaEq} (${txtFaltanEq})
${generarHTMLAsesor(eq.lider)}
`; // Filas de los asesores eq.asesores.forEach(as => { html += generarHTMLAsesor(as); }); html += `
`; divSeccion.innerHTML = html; container.appendChild(divSeccion); }); // Contabilizar turnos y total global en tiempo real Object.keys(estrellasData).forEach(nombre => { const historial = estrellasData[nombre] || []; const esAM = nombre.toUpperCase().includes('(AM)'); const esPM = nombre.toUpperCase().includes('(PM)'); totalEstrellasGeneral += historial.length; historial.forEach(campana => { if (esPM) { if (campana === 'claro') pmClaro++; if (campana === 'wow') pmWow++; if (campana === 'movistar' || campana === 'movistar_hogar') pmMovistarHogar++; if (campana === 'movistar_movil') pmMovistarMovil++; } else { if (campana === 'claro') amClaro++; if (campana === 'wow') amWow++; if (campana === 'movistar' || campana === 'movistar_hogar') amMovistarHogar++; if (campana === 'movistar_movil') amMovistarMovil++; } }); }); // Actualizar Contadores DOM const metaGen = metasData.general || 28; document.getElementById('metaGeneralDisplay').innerText = metaGen; document.getElementById('totalEstrellasGeneral').innerText = totalEstrellasGeneral; document.getElementById('contadorClaroAM').innerText = amClaro; document.getElementById('contadorWowAM').innerText = amWow; document.getElementById('contadorMovistarHogarAM').innerText = amMovistarHogar; document.getElementById('contadorMovistarMovilAM').innerText = amMovistarMovil; document.getElementById('contadorClaroPM').innerText = pmClaro; document.getElementById('contadorWowPM').innerText = pmWow; document.getElementById('contadorMovistarHogarPM').innerText = pmMovistarHogar; document.getElementById('contadorMovistarMovilPM').innerText = pmMovistarMovil; const faltantes = metaGen - totalEstrellasGeneral; const metaContador = document.getElementById('metaPizzaContador'); if (faltantes > 0) { metaContador.innerText = `Faltan ${faltantes}`; metaContador.style.color = "#FFB6C1"; metaContador.style.textShadow = "none"; metaLogradaPreviamente = false; } else { metaContador.innerText = "¡LOGRADO! 🎀💖"; metaContador.style.color = "#FF1493"; metaContador.style.textShadow = "0 0 10px #FF1493"; if (!metaLogradaPreviamente) { metaLogradaPreviamente = true; lluviaDeMonos(); } } }
